cmake_minimum_required(VERSION 3.20)
project(SnakeGame VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER ${config} CONFIG_UPPER)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CONFIG_UPPER} "${CMAKE_BINARY_DIR}/bin")
endforeach()
find_package(raylib 5.5 EXACT QUIET)
if(NOT raylib_FOUND)
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib
        URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz
        URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(raylib)
endif()
add_executable(SnakeGame src/main.cpp src/game.cpp src/board.cpp src/snake.cpp src/apple.cpp src/ui.cpp src/animation.cpp src/audio.cpp)
target_include_directories(SnakeGame PRIVATE include)
target_link_libraries(SnakeGame PRIVATE raylib)
add_custom_command(TARGET SnakeGame POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:SnakeGame>/assets")
if(MINGW)
    target_link_options(SnakeGame PRIVATE -static -mwindows)
endif()
if(MSVC)
    target_compile_options(SnakeGame PRIVATE /W4)
else()
    target_compile_options(SnakeGame PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(SnakeTests tests/logic_tests.cpp src/snake.cpp src/apple.cpp src/game.cpp src/board.cpp src/animation.cpp src/audio.cpp src/ui.cpp)
    target_include_directories(SnakeTests PRIVATE include)
    target_link_libraries(SnakeTests PRIVATE raylib)
    if(MINGW)
        target_link_options(SnakeTests PRIVATE -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME snake_logic COMMAND SnakeTests)
    add_executable(SnakeRuntimeTests tests/runtime_tests.cpp src/snake.cpp src/apple.cpp src/game.cpp src/board.cpp src/animation.cpp src/audio.cpp src/ui.cpp)
    target_include_directories(SnakeRuntimeTests PRIVATE include)
    target_link_libraries(SnakeRuntimeTests PRIVATE raylib)
    if(MINGW)
        target_link_options(SnakeRuntimeTests PRIVATE -static-libgcc -static-libstdc++)
    endif()
endif()
